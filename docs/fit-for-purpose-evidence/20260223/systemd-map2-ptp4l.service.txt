[Unit]
Description=MAP2 PTP4L gPTP Time Synchronization
Documentation=https://github.com/matthewmackes/map2-audio
ConditionPathExists=/etc/map2/avb-enabled
PartOf=map2-avb.target
After=network-online.target
Wants=network-online.target
Before=map2-backend.service

[Service]
Type=simple
ExecStart=/usr/sbin/ptp4l -f /etc/ptp4l.conf -m
Restart=on-failure
RestartSec=5s
# Note: No Requires= to prevent blocking boot if unavailable
# ptp4l output goes to journal for monitoring

[Install]
WantedBy=map2-avb.target
