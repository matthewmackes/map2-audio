## /etc/sysctl.d/91-map2-audio-rt.conf
# MAP2 Audio Platform - Realtime Scheduling Budget
# File: /etc/sysctl.d/91-map2-audio-rt.conf
#
# Description:
#   Configure realtime scheduling budget for low-latency audio processing.
#   Ensures audio thread gets guaranteed CPU time (2.95 sec out of 3 sec).
#   
# Applies to: AUDIO mode, ALL-IN-ONE mode
# Impact: Prevents starvation of audio thread if other RT threads created

# Realtime scheduling budget (98% of RT CPU time)
kernel.sched_rt_runtime_us = 2950000
kernel.sched_rt_period_us = 3000000

# Deadline scheduler tuning (if using SCHED_DEADLINE)
kernel.sched_deadline_period_us = 3000000
kernel.sched_deadline_runtime_us = 2950000

# Fair scheduler tuning for housekeeping cores
kernel.sched_autogroup_enabled = 0
## /etc/sysctl.d/92-map2-audio-thp.conf
# MAP2 Audio Platform - Memory & THP Management
# File: /etc/sysctl.d/92-map2-audio-thp.conf
#
# Description:
#   Disable Transparent Huge Pages (THP) completely for low-latency audio.
#   THP compaction can cause 1–5 ms latency spikes even when compaction_proactiveness=0.
#   Fixed page size (4 KB) is more predictable for realtime workloads.
#
# Applies to: AUDIO mode, ALL-IN-ONE mode
# Impact: Eliminates compaction pauses; reduces latency variance by 0.5–1.0 ms

# Disable THP completely
vm.transparent_hugepage = never
mm.transparent_hugepage.enabled = never
mm.transparent_hugepage.defrag = never
mm.transparent_hugepage.khugepaged.enabled = 0

# Page-level memory tuning
vm.page-cluster = 0                       # Read 1 page at a time (no readahead)
vm.overcommit_ratio = 50                  # Conservative memory commitment
vm.max_map_count = 262144                 # Allow more memory mappings

# Disable KSM (Kernel Samepage Merging) to avoid scanning/merging overhead
kernel.sched_migration_cost_ns = 500000   # Increase migration threshold
## /etc/sysctl.d/93-map2-audio-swappiness.conf
# MAP2 Audio Platform - Memory Swappiness & OOM
# File: /etc/sysctl.d/93-map2-audio-swappiness.conf
#
# Description:
#   Disable swapping and aggressive memory management for audio nodes.
#   Swapping to disk causes 10–100+ ms latency spikes (catastrophic for <3ms target).
#   Configure OOM killer to be predictable.
#
# Applies to: AUDIO mode, ALL-IN-ONE mode
# Impact: Prevents swap-induced latency; guarantees RT execution

# CRITICAL: Disable swap completely
vm.swappiness = 0                         # Never swap (prefer OOM kill)

# Memory pressure tuning
vm.oom_kill_allocating_task = 1           # Kill offending task immediately
vm.oom_dump_tasks = 0                     # Don't dump task list (saves time)
vm.panic_on_oom = 0                       # Don't panic; let OOM killer work

# Dirty page writeback (async I/O)
vm.dirty_ratio = 20                       # Bytes → disk at 20% dirtiness
vm.dirty_background_ratio = 10            # BG writeback at 10%
vm.dirty_expire_centisecs = 3000          # Expire after 30 sec

# Zone reclaim mode (NUMA-aware, disabled for predictability)
vm.zone_reclaim_mode = 0                  # Don't reclaim pages from remote NUMA

# Memory compaction
vm.compact_memory = 0                     # Manual trigger only
vm.compaction_proactiveness = 0           # Disable proactive compaction
## /etc/sysctl.d/94-map2-audio-watchdog.conf
# MAP2 Audio Platform - Kernel Watchdog & NMI Tuning
# File: /etc/sysctl.d/94-map2-audio-watchdog.conf
#
# Description:
#   Disable NMI watchdog and soft lockup detection.
#   These can steal ~1% CPU and cause unpredictable latency spikes.
#
# Applies to: AUDIO mode
# Impact: Frees CPU; reduces ~1% NMI overhead

# Disable NMI watchdog (costs ~1% CPU, periodic interrupts)
kernel.nmi_watchdog = 0

# Disable soft lockup detection
kernel.softlockup_panic = 0
kernel.softlockup_panic_all_cpu = 0

# Disable hung task detector
kernel.hung_task_timeout_secs = 0
kernel.hung_task_panic = 0

# Disable panic on kernel oops (let system continue)
kernel.panic_on_oops = 0
kernel.panic_on_unrecovered_nmi = 0

# Printk settings (reduce logging overhead)
kernel.printk_ratelimit = 0               # Disable rate limiting (or set high)
kernel.printk_ratelimit_burst = 1
